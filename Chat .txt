/* =========================================================

const $  = (s, r = document) => r.querySelector(s);
const $$ = (s, r = document) => [...r.querySelectorAll(s)];
const esc = s => String(s).replace(/[&<>"']/g, c =>
  ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const uid = () => Date.now().toString(36) + Math.random().toString(36).slice(2, 7);
const chatId = (a, b) => [a, b].sort().join('__');
const fmt = t => new Date(t).toLocaleTimeString('fa-IR', { hour: '2-digit', minute: '2-digit' });
const MAX_FILE = 2 * 1024 * 1024; // ۲ مگابایت

const bc = ('BroadcastChannel' in window) ? new BroadcastChannel('chatroom_sync') : null;

const store = {
  get(k, def) { try { const v = localStorage.getItem(k); return v ? JSON.parse(v) : def; } catch (e) { return def; } },
  set(k, v) { localStorage.setItem(k, JSON.stringify(v)); if (bc) bc.postMessage({ type: 'sync' }); }
};

/* داده‌ها */
const users    = () => store.get('cr_users', {});
const setUsers = v  => store.set('cr_users', v);
const reqs     = () => store.get('cr_reqs', []);
const setReqs  = v  => store.set('cr_reqs', v);
const allMsgs  = () => store.get('cr_msgs', {});
const setMsgs  = v  => store.set('cr_msgs', v);
const seenMap  = () => store.get('cr_seen', {});

/* =========================================================
   وضعیت
   ========================================================= */
let state = { me: null, chat: null, tab: 'friends', mode: 'login' };
let seenReqIds = new Set();
let knownMsgCount = {};
let booted = false;

/* =========================================================
   Toast
   ========================================================= */
let toastTimer;
function toast(msg, kind = '') {
  const el = $('#toast');
  el.textContent = msg;
  el.className = 'toast show ' + kind;
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => { el.className = 'toast ' + kind; }, 2800);
}

/* =========================================================
   احراز هویت
   ========================================================= */
function genCode() {
  const u = users();
  const used = new Set(Object.values(u).map(x => x.code));
  let c;
  do { c = String(Math.floor(100000 + Math.random() * 900000)); } while (used.has(c));
  return c;
}

$$('.switch button').forEach(b => b.onclick = () => {
  $$('.switch button').forEach(x => x.classList.remove('active'));
  b.classList.add('active');
  state.mode = b.dataset.mode;
  $('#authBtn').textContent = state.mode === 'login' ? 'ورود' : 'ساخت حساب';
});

function doAuth() {
  const name = $('#authUser').value.trim();
  const pass = $('#authPass').value;
  if (name.length < 3)  return toast('نام کاربری حداقل ۳ کاراکتر باشه', 'bad');
  if (pass.length < 4)  return toast('رمز عبور حداقل ۴ کاراکتر باشه', 'bad');

  const u = users();

  if (state.mode === 'register') {
    if (u[name]) return toast('این نام کاربری قبلاً گرفته شده', 'bad');
    u[name] = { pass, code: genCode(), created: Date.now() };
    setUsers(u);
    toast('حساب ساخته شد 🎉 کد تو: ' + u[name].code, 'ok');
  } else {
    if (!u[name])               return toast('کاربری با این نام پیدا نشد', 'bad');
    if (u[name].pass !== pass)  return toast('رمز عبور اشتباهه', 'bad');
  }

  sessionStorage.setItem('cr_session', name);
  state.me = name;
  enterApp();
}

$('#authBtn').onclick = doAuth;
$('#authPass').onkeydown = e => { if (e.key === 'Enter') doAuth(); };

  $('#appScreen').classList.remove('hidden');
  state.chat = null;
  seenReqIds = new Set(reqs().filter(r => r.to === state.me).map(r => r.id));
  knownMsgCount = {};
  renderSidebar();
  refresh();
  booted = true;
}

  sessionStorage.removeItem('cr_session');
  state.me = null;
  state.chat = null;
  booted = false;
  $('#appScreen').classList.add('hidden');
  $('#authScreen').classList.remove('hidden');
  $('#authPass').value = '';
}

$$('.tab').forEach(t => t.onclick = () => switchTab(t.dataset.tab));

function switchTab(tab) {
  state.tab = tab;
  $$('.tab').forEach(x => x.classList.toggle('active', x.dataset.tab === tab));
  $$('.panel').forEach(p => p.classList.toggle('active', p.id === 'panel-' + tab));

function friendsOf(user) {
  return reqs()
    .filter(r => r.status === 'accepted' && (r.from === user || r.to === user))
    .map(r => (r.from === user ? r.to : r.from));
}

function unreadCount(friend) {
  const id = chatId(state.me, friend);
  const last = seenMap()[state.me + '|' + id] || 0;
  return (allMsgs()[id] || []).filter(m => m.from === friend && m.t > last).length;
}

function markSeen(friend) {
  const s = seenMap();
  s[state.me + '|' + chatId(state.me, friend)] = Date.now();
  localStorage.setItem('cr_seen', JSON.stringify(s)); // بدون broadcast
}

function renderSidebar() {
  if (!state.me) return;
  const me = users()[state.me] || {};
  $('#meName').textContent = state.me;
  $('#meCode').textContent = me.code || '—';
  $('#meAvatar').textContent = state.me[0].toUpperCase();

  renderFriends();
  renderRequests();
  updateBadge();
}

function updateBadge() {
  const n = reqs().filter(r => r.to === state.me && r.status === 'pending').length;
  const b = $('#reqBadge');
  b.textContent = n;
  b.classList.toggle('hidden', n === 0);
}

function renderFriends() {
  const box = $('#panel-friends');
  const list = friendsOf(state.me);

  if (!list.length) {
    box.innerHTML = `<div class="empty">هنوز دوستی نداری 👥<br><small>از تب «افزودن» با کد شروع کن</small></div>`;
    return;
  }

  const msgs = allMsgs();

  box.innerHTML = list.map(f => {
    const id = chatId(state.me, f);
    const arr = msgs[id] || [];
    const last = arr[arr.length - 1];
    const un = unreadCount(f);
    let preview = 'شروع گفتگو…';
    if (last) {
      preview = last.type === 'text' ? last.data
              : last.type === 'image' ? '🖼️ عکس'
              : '🎬 ویدیو';
      if (last.from === state.me) preview = 'شما: ' + preview;
    }
    return `<div class="item ${state.chat === f ? 'active' : ''}" data-friend="${esc(f)}">
      <div class="avatar sm">${esc(f[0].toUpperCase())}</div>
      <div class="item-body">
        <div class="item-top">
          <span class="item-name">${esc(f)}</span>
          <span class="item-time">${last ? fmt(last.t) : ''}</span>
        </div>
        <div class="item-sub">
          <span style="overflow:hidden;text-overflow:ellipsis">${esc(String(preview).slice(0, 38))}</span>
          ${un ? `<span class="dot">${un}</span>` : ''}
        </div>
      </div>
    </div>`;
  }).join('');
}

$('#panel-friends').onclick = e => {

function renderRequests() {
  const box = $('#panel-requests');
  const all = reqs();
  const incoming = all.filter(r => r.to === state.me).sort((a, b) => b.t - a.t);
  const outgoing = all.filter(r => r.from === state.me).sort((a, b) => b.t - a.t);

  let html = '';
  if (!incoming.length && !outgoing.length) {
    box.innerHTML = `<div class="empty">درخواستی نداری 📭</div>`;
    return;
  }

  if (incoming.length) {
    html += `<div class="sec-title">دریافتی</div>`;
    html += incoming.map(r => `
      <div class="req">
        <div class="avatar sm">${esc(r.from[0].toUpperCase())}</div>
        <div class="req-body">
          <div class="item-name">${esc(r.from)}</div>
          <div class="item-sub">${
            r.status === 'pending'  ? 'درخواست دوستی فرستاد' :
            r.status === 'accepted' ? 'قبول شد ✅' : 'رد شد ✖'
          }</div>
        </div>
        ${r.status === 'pending' ? `
          <div class="req-actions">
            <button class="ok" data-accept="${r.id}">قبول</button>
            <button class="no" data-reject="${r.id}">رد</button>
          </div>` : ''}
      </div>`).join('');
  }

  if (outgoing.length) {
    html += `<div class="sec-title">فرستاده‌شده</div>`;
    html += outgoing.map(r => `
      <div class="req">
        <div class="avatar sm">${esc(r.to[0].toUpperCase())}</div>
        <div class="req-body">
          <div class="item-name">${esc(r.to)}</div>
          <div class="item-sub">${
            r.status === 'pending'  ? 'در انتظار تأیید ⏳' :
            r.status === 'accepted' ? 'قبول شد ✅' : 'رد شد ✖'
          }</div>
        </div>
      </div>`).join('');
  }

  box.innerHTML = html;
}

$('#panel-requests').onclick = e => {
  const acc = e.target.closest('[data-accept]');
  const rej = e.target.closest('[data-reject]');
  if (acc) respondRequest(acc.dataset.accept, 'accepted');
  if (rej) respondRequest(rej.dataset.reject, 'rejected');
};

function respondRequest(id, status) {
  const r = reqs();
  const req = r.find(x => x.id === id);
  if (!req) return;
  req.status = status;
  setReqs(r);
  renderSidebar();
  if (status === 'accepted') {
    switchTab('friends');
    toast('دوست شدید! 🎉 حالا می‌تونید چت کنید', 'ok');
    openChat(req.from);
  } else {
    toast('درخواست رد شد');
  }
}

$('#friendCode').onkeydown = e => { if (e.key === 'Enter') sendInvite(); };

function sendInvite() {
  const code = $('#friendCode').value.trim();
  if (!code) return toast('کد دوستت رو وارد کن', 'bad');

  const u = users();
  const target = Object.keys(u).find(k => u[k].code === code);
  if (!target)             return toast('کدی با این شماره پیدا نشد', 'bad');
  if (target === state.me) return toast('نمی‌تونی خودت رو اضافه کنی 😄', 'bad');

  const r = reqs();
  if (r.some(x => (x.from === state.me && x.to === target) || (x.from === target && x.to === state.me)))
    return toast('قبلاً بین شما درخواست ثبت شده', 'bad');

  r.push({ id: uid(), from: state.me, to: target, status: 'pending', t: Date.now() });
  setReqs(r);

  $('#friendCode').value = '';
  renderSidebar();
  toast('دعوت فرستاده شد ✅ منتظر تأیید ' + target + ' باش', 'ok');
}

function openChat(friend) {
  state.chat = friend;
  markSeen(friend);
  buildChatUI();

}

}

function buildChatUI() {

  $('#btnVid').onclick = () => { const i = $('#fileVid'); i.value = ''; i.click(); };

  $('#fileVid').onchange = e => handleFile(e.target.files[0], 'video');
  $('#msgInput').focus();

function renderMessages() {
  const box = $('#msgs');
  if (!box || !state.chat) return;

  const id = chatId(state.me, state.chat);
  const list = allMsgs()[id] || [];

  if (!list.length) {
    box.innerHTML = `<div class="empty-chat">شروع گفتگو کن… ✨</div>`;
    return;
  }

  box.innerHTML = list.map(m => {
    const mine = m.from === state.me;
    let body = '';
    if (m.type === 'text') {
      body = `<div>${esc(m.data)}</div>`;
    } else if (m.type === 'image') {
      body = `<img class="media" src="${m.data}" onclick="openLightbox(this.src)">`;
    } else if (m.type === 'video') {
      body = `<video class="media" src="${m.data}" controls preload="metadata"></video>`;
    }
    return `<div class="msg ${mine ? 'mine' : 'theirs'}">
      ${body}
      <div class="time">${fmt(m.t)}</div>
    </div>`;
  }).join('');

  box.scrollTop = box.scrollHeight;
}

function sendText() {
  const inp = $('#msgInput');
  const v = inp.value.trim();
  if (!v) return;
  inp.value = '';
  pushMessage('text', v);
}

function pushMessage(type, data) {
  if (!state.chat) return;
  const id = chatId(state.me, state.chat);
  const all = allMsgs();
  if (!all[id]) all[id] = [];
  all[id].push({ id: uid(), from: state.me, type, data, t: Date.now() });

  try {
    setMsgs(all);
  } catch (e) {
    toast('حافظه پر شده! پیام‌های قدیمی رو پاک کن', 'bad');
    return;
  }

  markSeen(state.chat);
  renderMessages();
  renderSidebar();
}

function handleFile(file, type) {
  if (!file) return;
  if (file.size > MAX_FILE)
    return toast('حجم فایل باید کمتر از ۲ مگابایت باشه', 'bad');

  const r = new FileReader();
  r.onload = () => pushMessage(type, r.result);
  r.onerror = () => toast('خوندن فایل ناموفق بود', 'bad');
  r.readAsDataURL(file);
}

function refresh() {
  if (!state.me || !booted) return;
  renderSidebar();
  if (state.chat) renderMessages();
  notifyNew();
}

function notifyNew() {
  // درخواست‌های جدید
  reqs().filter(r => r.to === state.me && r.status === 'pending').forEach(r => {
    if (!seenReqIds.has(r.id)) {
      seenReqIds.add(r.id);
      toast(`🔔 ${r.from} بهت درخواست دوستی داد`, 'ok');
    }
  });

  // پیام‌های جدید
  const msgs = allMsgs();
  friendsOf(state.me).forEach(f => {
    const id = chatId(state.me, f);
    const n = (msgs[id] || []).filter(m => m.from === f).length;
    if (knownMsgCount[id] !== undefined && n > knownMsgCount[id] && f !== state.chat) {
      toast(`💬 پیام جدید از ${f}`, 'ok');
    }
    knownMsgCount[id] = n;
  });

  // پیام‌های چت باز
  if (state.chat) {
    const id = chatId(state.me, state.chat);
    const n = (msgs[id] || []).length;
    if (knownMsgCount['open_' + id] !== undefined && n > knownMsgCount['open_' + id]) {
      markSeen(state.chat);
    }
    knownMsgCount['open_' + id] = n;
  }
}
